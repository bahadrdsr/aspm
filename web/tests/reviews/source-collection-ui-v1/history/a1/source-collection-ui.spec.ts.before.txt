async function exactCollection(page: Page, value: UICollection) {
  const panel = collectionDetails(page);
  await expect(collectionStatus(page)).toContainText(new RegExp(`\\b${value.state}\\b`, "i"));
  for (const text of [value.id, value.repository]) await expect(panel).toContainText(text);
  await expect(panel).toContainText(new RegExp(`\\b${value.recordCount}\\s+records?\\b|record count\\s*:?\\s*${value.recordCount}\\b`, "i"));
  for (const time of [value.createdAt, value.collectedAt, value.completedAt].filter((stamp): stamp is string => stamp !== null)) {
    await expect(panel.locator(`time[datetime="${time}"]`).first()).toBeVisible();
  }
  if (value.complete) await expect(panel).toContainText(/selected (?:feeds|feed).*exhaust|selected.*complete|complete.*selected/i);
  else await expect(panel).toContainText(/incomplete|not complete|complete\s*:\s*false|not (?:yet )?(?:collected|finished)|queued|collecting|blocked|failed|partial/i);
  for (const gap of value.gaps) await expect(panel).toContainText(gap);
  if (value.failure) {
    await expect(panel).toContainText(value.failure.code);
    if (value.failure.nativeCode) await expect(panel).toContainText(value.failure.nativeCode);
    await expect(panel).toContainText(new RegExp(`HTTP(?: status)?\\s*:?\\s*${value.failure.httpStatus}\\b`, "i"));
    if (value.failure.retryAfterSeconds) await expect(panel).toContainText(/retry.after\s*:?\\s*7\s*(?:s\b|seconds?)/i);
    await expect(panel).toContainText(/no automatic retr|not (?:automatically )?retried|retryable\s*:\s*false/i);
  }
  if (value.assetId) {
    await expect(panel).toContainText(value.assetId);
    for (const link of await panel.getByRole("link", { name: /asset/i })
      .or(panel.getByRole("link", { name: value.assetId, exact: true })).all()) {
      const target = new URL(await link.getAttribute("href") ?? "", page.url());
      expect(target.hash, "An asset reference is not permission to invent a direct-detail/filter UI route.").toBe("#/assets");
    }
  }
  await expect(panel.getByText(/^Scan passed$|^Verified$|^Resolved$|^Normalized finding$/i)).toHaveCount(0);

async function rawEvidence(page: Page, sources: SourcesUIAPI, value: UIRecord) {
  const raw = sources.records.get(value.id)!.raw, text = raw.toString("utf8"), viewer = evidence(page);
  const pre = viewer.locator("pre");
  await expect(pre).toHaveCount(1); await expect(pre).toBeVisible();
  const dom = await pre.textContent();
  expect(dom === text || dom === text.replaceAll("\r\n", "\n"), "Raw bytes must not be JSON-reserialized, trimmed or interpreted as markup.").toBe(true);
  await expect(pre.locator("a, script, iframe, img, svg, b, em")).toHaveCount(0);
  if (dom !== text) {
    const download = page.waitForEvent("download");
    await viewer.getByRole("button", { name: /^Download(?: raw)? evidence$|^Download raw record$/i })
      .or(viewer.getByRole("link", { name: /^Download(?: raw)? evidence$|^Download raw record$/i })).click();
    const stream = await (await download).createReadStream();
    if (!stream) throw new Error("No exact-byte evidence download.");
    const chunks: Buffer[] = []; for await (const chunk of stream) chunks.push(Buffer.from(chunk));
    expect(Buffer.concat(chunks).equals(raw), "A line-normalized display still needs an exact raw-byte download.").toBe(true);
  }
  const call = sources.calls("GET", evidencePath(value.collectionId, value.id)).at(-1)!;
  expect(call.rawSHA256).toBe(value.evidence.sha256);
  expect(call.rawBytes).toBe(value.evidence.sizeBytes);
}

async function noDecorativeMovement(page: Page) {
  const moving = await page.evaluate(async () => {
    const found = new Set<string>();
    for (let sample = 0; sample < 6; sample++) {
      for (const animation of document.getAnimations()) {
        if (animation.playState !== "running" || !(animation.effect instanceof KeyframeEffect)) continue;
        if (animation.effect.getTiming().iterations === Infinity) found.add("continuous");
        const frames = animation.effect.getKeyframes() as Array<Record<string, unknown>>;
        for (const key of ["transform", "translate", "scale", "rotate", "top", "left"]) {
          if (new Set(frames.map((frame) => frame[key]).filter((value) => value !== undefined).map(String)).size > 1) found.add(key);
        }
      }
      await new Promise<void>((resolve) => requestAnimationFrame(() => resolve()));
    }
    return [...found];
  });
  expect(moving).toEqual([]);
}
