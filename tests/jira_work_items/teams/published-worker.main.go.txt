package main

import (
	"context"
	"crypto/tls"
	"crypto/x509"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"net"
	"net/http"
	"net/url"
	"os"
	"regexp"
	"time"

	"github.com/bahadrdsr/aspm/internal/app"
)

type publishedWorkerInput struct {
	DatabaseURL, Schema, WorkerID, JiraOrigin, SlackOrigin string
	EncryptionKey, RootDER                                []byte
}

func init() {
	if os.Getenv("ASPM_TEAMS_PUBLISHED_WORKER") != "1" {
		return
	}
	if err := publishedWorkerStep(); err != nil {
		fmt.Fprintln(os.Stderr, "owned published V10 worker failed; private details withheld")
		os.Exit(1)
	}
	os.Exit(0)
}

func publishedWorkerStep() error {
	var in publishedWorkerInput
	decoder := json.NewDecoder(io.LimitReader(os.Stdin, 64<<10))
	decoder.DisallowUnknownFields()
	if decoder.Decode(&in) != nil || !regexp.MustCompile(`^jira_items_[a-f0-9]{24}$`).MatchString(in.Schema) ||
		len(in.EncryptionKey) != 32 {
		return errors.New("unowned published worker input")
	}
	database, err := url.Parse(in.DatabaseURL)
	if err != nil || database.Hostname() != "127.0.0.1" || database.Port() != "15432" || database.User == nil {
		return errors.New("unapproved published worker PG boundary")
	}
	allowed := map[string]bool{}
	for _, value := range []string{in.JiraOrigin, in.SlackOrigin} {
		u, err := url.Parse(value)
		if err != nil || u.Scheme != "https" || u.Hostname() != "127.0.0.1" || u.Port() == "" ||
			u.Path != "" || u.RawQuery != "" || u.Fragment != "" || u.User != nil {
			return errors.New("unapproved published native origin")
		}
		allowed[u.Host] = true
	}
	root, err := x509.ParseCertificate(in.RootDER)
	if err != nil {
		return errors.New("invalid owned native TLS root")
	}
	roots := x509.NewCertPool()
	roots.AddCert(root)
	blocked := &http.Transport{Proxy: nil, DialContext: func(context.Context, string, string) (net.Conn, error) {
		return nil, errors.New("published worker ambient HTTP forbidden")
	}}
	http.DefaultTransport = blocked
	defer blocked.CloseIdleConnections()
	transport := &http.Transport{Proxy: nil, TLSClientConfig: &tls.Config{RootCAs: roots, MinVersion: tls.VersionTLS12}}
	transport.DialContext = func(ctx context.Context, network, address string) (net.Conn, error) {
		if !allowed[address] {
			return nil, errors.New("published worker unapproved native destination")
		}
		return (&net.Dialer{Timeout: time.Second}).DialContext(ctx, network, address)
	}
	defer transport.CloseIdleConnections()
	ctx, cancel := context.WithTimeout(context.Background(), 8*time.Second)
	defer cancel()
	var queries queryBudget
	worker, err := app.OpenDeliveryWorker(ctx, app.DeliveryWorkerConfig{
		Database: app.DatabaseConfig{DatabaseURL: in.DatabaseURL, Schema: in.Schema,
			ApplicationName: "owned-published-v10-worker", MaxConnections: 1, QueryTracer: &queries, LogOutput: io.Discard},
		EncryptionKey: in.EncryptionKey, WorkerID: in.WorkerID, LeaseDuration: 4*time.Second,
		SlackEndpoint: in.SlackOrigin, Client: &http.Client{Transport: transport, Timeout: 3*time.Second,
			CheckRedirect: func(*http.Request, []*http.Request) error { return http.ErrUseLastResponse }},
	})
	if err != nil {
		return errors.New("published worker open failed")
	}
	defer worker.Close()
	processed, err := worker.ProcessNext(ctx)
	if err != nil || !processed {
		return errors.New("published bounded delivery step failed")
	}
	return json.NewEncoder(os.Stdout).Encode(map[string]any{"queries": queries.count.Load(), "steps": 1})
}
